int pinPOT = A0;
int pinLED = 3;
int BRILLO = 0;

void setup() {
  pinMode(A0, INPUT);
  pinMode(3, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  BRILLO = analogRead(A0);
  Serial.println(BRILLO); // Muestra el valor original (0 a 1023)
  
  BRILLO = (BRILLO / 4);  // Convierte de 0-1023 a 0-255
  analogWrite(3, BRILLO);
  Serial.println(BRILLO); // Muestra el valor mapeado para el LED (0 a 255)
  
  delay(10); // Pequeña pausa para mejorar el rendimiento de la simulación
}
